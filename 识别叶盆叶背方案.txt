一、项目概述
1.1 目标
开发一个基于Open CASCADE Technology（OCCT）的自动化程序，能够从STEP或IGES格式的叶片CAD模型中，自动识别并提取出叶盆面（压力面） 和叶背面（吸力面） 两个独立的NURBS曲面，同时自动排除模型中存在的其他非叶片几何面（如安装座、加强筋、流道等干扰面）。
1.2 技术基础
•	平台：基于OCCT几何内核（C++）
•	输入：STEP（.stp/.step）或IGES（.igs/.iges）格式文件
•	输出：两个独立的NURBS曲面（叶盆、叶背）

三、详细技术实现
模块1：文件读取与模型加载
3.1.1 STEP文件读取
使用STEPControl_Reader类读取STEP文件：
cpp
#include <STEPControl_Reader.hxx>
#include <IFSelect_ReturnStatus.hxx>
#include <TopoDS_Shape.hxx>

TopoDS_Shape ReadSTEPFile(const std::string& filename)
{
    STEPControl_Reader reader;
    IFSelect_ReturnStatus status = reader.ReadFile(filename.c_str());
    
    if (status != IFSelect_RetDone) {
        // 文件读取失败处理
        return TopoDS_Shape();
    }
    
    // 检查可转换的根实体数量
    Standard_Integer nbRoots = reader.NbRootsForTransfer();
    if (nbRoots == 0) {
        return TopoDS_Shape();
    }
    
    // 转换所有根实体
    reader.TransferRoots();
    
    // 获取转换后的形状（通常取第一个）
    return reader.OneShape();
}
3.1.2 IGES文件读取
使用IGESControl_Reader类读取IGES文件：

#include <IGESControl_Reader.hxx>

TopoDS_Shape ReadIGESFile(const std::string& filename)
{
    IGESControl_Reader reader;
    IFSelect_ReturnStatus status = reader.ReadFile(filename.c_str());
    
    if (status != IFSelect_RetDone) {
        return TopoDS_Shape();

}
3.1.3 统一入口函数
根据文件扩展名自动选择读取器：
cpp
TopoDS_Shape ReadCADFile(const std::string& filename)
{
    std::string ext = filename.substr(filename.find_last_of(".") + 1);
    std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
    
    if (ext == "stp" || ext == "step") {
        return ReadSTEPFile(filename);
    } else if (ext == "igs" || ext == "iges") {
        return ReadIGESFile(filename);
    } else {
        // 不支持的文件格式
        return TopoDS_Shape();
    }
}

模块4：截面生成与叶盆/叶背识别
这是整个方案的核心模块，通过生成叶片在不同高度上的截面线，然后基于曲率特征将截面线分割为叶盆、叶背、前缘、后缘四部分。
3.3.1 生成截面线
使用BRepAlgoAPI_Section计算叶片与切割平面的交线：
cpp
#include <BRepAlgoAPI_Section.hxx>
#include <gp_Pln.hxx>
#include <BRepBuilderAPI_MakeEdge.hxx>
#include <TopoDS_Compound.hxx>
#include <BRep_Builder.hxx>

std::vector<TopoDS_Edge> GetSectionCurves(const TopoDS_Shape& shape, 
                                           double zHeight)
{
    // 创建切割平面（法向为Z轴）
    gp_Pln cuttingPlane(gp_Pnt(0, 0, zHeight), gp_Dir(0, 0, 1));
    
    // 计算截面
    BRepAlgoAPI_Section section(shape, cuttingPlane, false);
    section.ComputePCurveOn1(true);
    section.Approximation(true);
    section.Build();
    
    if (!section.IsDone()) {
        return std::vector<TopoDS_Edge>();
    }
    
    // 提取截面边
    TopoDS_Shape result = section.Shape();
    std::vector<TopoDS_Edge> edges;
    
    for (TopExp_Explorer exp(result, TopAbs_EDGE); exp.More(); exp.Next()) {
        TopoDS_Edge edge = TopoDS::Edge(exp.Current());
        edges.push_back(edge);
    }
    
    return edges;
}

3.3.3 基于曲率的分界点识别
叶盆和叶背的曲率特征明显不同，前缘和后缘区域曲率较大（曲率半径小），而叶盆和叶背区域曲率相对平缓。利用这一特征可以找到四个区域的分界点：
cpp
#include <BRepAdaptor_Curve.hxx>
#include <GeomLProp_CLProps.hxx>
#include <GCPnts_UniformAbscissa.hxx>
#include <vector>

struct SectionPoint {
    gp_Pnt point;
    double param;
    double curvature;
};

std::vector<SectionPoint> SampleCurvePoints(const TopoDS_Edge& edge, 
                                             int numSamples)
{
    std::vector<SectionPoint> points;
    BRepAdaptor_Curve adaptor(edge);
    
    double first = adaptor.FirstParameter();
    double last = adaptor.LastParameter();
    
    for (int i = 0; i <= numSamples; i++) {
        double t = first + (last - first) * i / numSamples;
        gp_Pnt p;
        gp_Vec d1, d2;
        adaptor.D2(t, p, d1, d2);
        
        // 计算曲率：k = |d1 × d2| / |d1|³
        double curvature = 0.0;
        double d1Mag = d1.Magnitude();
        if (d1Mag > 1e-10) {
            gp_Vec cross = d1.Crossed(d2);
            curvature = cross.Magnitude() / (d1Mag * d1Mag * d1Mag);
        }
        
        SectionPoint pt;
        pt.point = p;
        pt.param = t;
        pt.curvature = curvature;
        points.push_back(pt);
    }
    
    return points;
}

// 识别曲率极值点（前后缘分界点）
std::vector<int> FindCurvatureExtremes(const std::vector<SectionPoint>& points)
{
    std::vector<int> extremes;
    // 找到曲率局部极大值点（对应前缘和后缘区域）
    for (size_t i = 1; i < points.size() - 1; i++) {
        if (points[i].curvature > points[i-1].curvature && 
            points[i].curvature > points[i+1].curvature) {
            extremes.push_back(i);
        }
    }
    return extremes;
}

模块5：目标曲面提取与输出
3.4.1 从截面点集重建NURBS曲面
将多个高度层上提取的叶盆/叶背截面点集，分别拟合成NURBS曲面：
cpp
#include <GeomAPI_PointsToBSplineSurface.hxx>
#include <TColgp_Array2OfPnt.hxx>

Handle(Geom_BSplineSurface) RebuildSurface(
    const std::vector<std::vector<gp_Pnt>>& sectionPoints)
{
    // sectionPoints: 多个高度层的截面点集
    // 每个截面点的数量应保持一致
    
    int numSections = sectionPoints.size();
    int numPointsPerSection = sectionPoints[0].size();
    
    TColgp_Array2OfPnt points(1, numPointsPerSection, 1, numSections);
    
    for (int j = 0; j < numSections; j++) {
        for (int i = 0; i < numPointsPerSection; i++) {
            points.SetValue(i + 1, j + 1, sectionPoints[j][i]);
        }
    }
    
    GeomAPI_PointsToBSplineSurface surfaceBuilder(points);
    return surfaceBuilder.Surface();
}

四、关键技术难点与解决方案
4.1 干扰面的自动识别与剔除
难点：叶片模型可能包含安装座、加强筋、流道面等非叶片几何。
解决方案：
1.	基于曲面类型过滤：只保留NURBS曲面（GeomAbs_BSplineSurface），排除平面、圆柱面等
2.	基于面积过滤：叶片主体面积通常较大，可设定面积阈值
3.	基于拓扑关系：分析面的邻接关系，叶片主体通常构成一个封闭或半封闭的壳体
4.	基于位置分布：叶片面沿叶高方向分布，干扰面通常在特定区域
4.2 前后缘的精确处理
难点：前缘和后缘是叶盆和叶背的过渡区域，分割精度直接影响曲面质量。
解决方案：
1.	采用曲率阈值法：设定曲率阈值，曲率超过阈值的区域为前缘/后缘
2.	采用曲率变化率：寻找曲率变化最剧烈的点作为分界点
3.	对分割后的点集进行平滑处理，确保边界光滑
4.3 多个高度层截面的自动生成
难点：需要自动确定叶片的高度范围和合适的截面层数。
解决方案：
1.	计算叶片模型的包围盒（BRepBndLib），获取Z方向范围
2.	根据叶片高度自动计算截面层数（如每隔5mm取一个截面）
3.	在叶根和叶尖区域适当加密截面
